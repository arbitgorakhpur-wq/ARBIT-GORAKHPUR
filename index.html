import pandas as pd import openpyxl from openpyxl.styles import Font, PatternFill, Alignment, Border, Side from openpyxl.utils import get_column_letter # 1. Purani Excel file load karein input_filename = 'New Microsoft Excel Worksheet_3.xlsx' # Aapki input file ka naam df_raw = pd.read_excel(input_filename, header=None) # 2. Naya Excel Workbook create karein wb = openpyxl.Workbook() ws = wb.active ws.title = "Price List" # Styling definitions header_fill = PatternFill(start_color="1F4E79", end_color="1F4E79", fill_type="solid") header_font = Font(name="Calibri", size=11, bold=True, color="FFFFFF") dealer_fill = PatternFill(start_color="D9E1F2", end_color="D9E1F2", fill_type="solid") dealer_font = Font(name="Calibri", size=12, bold=True, color="000000") cat_fill = PatternFill(start_color="F2F2F2", end_color="F2F2F2", fill_type="solid") cat_font = Font(name="Calibri", size=11, bold=True, color="1F4E79") thin_border = Border( left=Side(style='thin', color='BFBFBF'), right=Side(style='thin', color='BFBFBF'), top=Side(style='thin', color='BFBFBF'), bottom=Side(style='thin', color='BFBFBF') ) # 3. Dealer ka Naam Header me merge karke add karein (Row 1) ws.merge_cells("A1:F1") ws["A1"] = df_raw.iloc[0, 0] # Dealer Name & Address ws["A1"].font = dealer_font ws["A1"].fill = dealer_fill ws["A1"].alignment = Alignment(horizontal="center", vertical="center") # 4. Table Headers set karein (Row 2) headers = ['Model Name', 'Ex-Showroom Price', 'RTO UP 53', 'Zero Dep Insurance', 'FASTag', '1 % TCS'] for col_num, header in enumerate(headers, 1): cell = ws.cell(row=2, column=col_num, value=header) cell.font = header_font cell.fill = header_fill cell.alignment = Alignment(horizontal="center", vertical="center") # 5. Data rows process karein current_row = 3 for i in range(2, len(df_raw)): row_val = df_raw.iloc[i].tolist() model_name = str(row_val[0]).strip() if pd.notna(row_val[0]) else '' if not model_name: continue # Agar Category Header ho (jaise NEW BREZZA, NEW ERTIGA) if pd.isna(row_val[1]): ws.merge_cells(start_row=current_row, start_column=1, end_row=current_row, end_column=6) cell = ws.cell(row=current_row, column=1, value=model_name) cell.font = cat_font cell.fill = cat_fill cell.alignment = Alignment(horizontal="left", vertical="center") else: # Car Variant Row ex_price = float(row_val[1]) rto = float(row_val[2]) if pd.notna(row_val[2]) else 0 ins = float(row_val[3]) if pd.notna(row_val[3]) else 0 fastag = float(row_val[4]) if pd.notna(row_val[4]) else 600 ws.cell(row=current_row, column=1, value=model_name) ws.cell(row=current_row, column=2, value=ex_price) ws.cell(row=current_row, column=3, value=rto) ws.cell(row=current_row, column=4, value=ins) ws.cell(row=current_row, column=5, value=fastag) # Excel Formula: Agar Ex-Showroom > 10,00,000 toh 1% TCS/TDS, nahi toh 0 tcs_formula = f'=IF(B{current_row}>1000000, B{current_row}*0.01, 0)' ws.cell(row=current_row, column=6, value=tcs_formula) # Currency Format ₹ for c in range(2, 7): ws.cell(row=current_row, column=c).number_format = '₹#,##0.00' # Borders for c in range(1, 7): ws.cell(row=current_row, column=c).border = thin_border current_row += 1 # Column Widths auto adjust karein for col in ws.columns: max_len = max(len(str(cell.value or '')) for cell in col) col_letter = get_column_letter(col[0].column) ws.column_dimensions[col_letter].width = max(max_len + 3, 15) # 6. File Save karein wb.save("Updated_Maruti_Price_List.xlsx") print("File successfully created: Updated_Maruti_Price_List.xlsx")
